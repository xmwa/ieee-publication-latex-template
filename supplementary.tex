% supplementary.tex: supplementary material, S-numbered so the main paper can cite "Fig. S1", "Eq. (S1)", "Table S-I", and "Section S-I" unambiguously.
% Shares preamble/authors.tex, bib/refs.bib, and figures/ with main.tex; its own sections and tables are in supplementary/.
% Revision marks: 1 = clean (default), 0 = marked; same switch as main.tex, and the command line wins over this line.
\ifdefined\CleanCopy\else\def\CleanCopy{1}\fi
\input{preamble/venue}
\documentclass[\PaperMode]{IEEEtran}

\usepackage{cite}
\usepackage{amsmath,amssymb}
\usepackage{graphicx}
\graphicspath{{figures/}}
\usepackage{booktabs}
\usepackage[hidelinks]{hyperref}

\input{preamble/authors}
\input{preamble/revision}

\hypersetup{
  pdftitle={Supplementary Material for \PaperTitle},
  pdfauthor={\PaperPdfAuthor},   % empty in anonymous mode
  pdfkeywords={\PaperKeywords},
}

% S-prefixed numbering (IEEE convention: sections and tables Roman, figures and equations arabic).
\renewcommand{\thesection}{S-\Roman{section}}
\renewcommand{\thetable}{S-\Roman{table}}
\renewcommand{\thefigure}{S\arabic{figure}}
\renewcommand{\theequation}{S\arabic{equation}}

% Wider number boxes in the table of contents: IEEEtran's are sized for "VIII", and "S-VIII" runs into the section title.
\makeatletter
\def\l@section#1#2{\addpenalty{\@secpenalty}\addvspace{1.0em plus 1pt}\@tempdima 3.7em \begingroup \parindent \z@ \rightskip \@pnumwidth \parfillskip-\@pnumwidth {\bfseries\leavevmode #1}\hfil\hbox to\@pnumwidth{\hss #2}\par\endgroup}
\def\l@subsection{\@dottedtocline{2}{1.0em}{4.9em}}
\makeatother

\begin{document}

\title{Supplementary Material for\\ ``\PaperTitle''}
\PaperAuthors
\PaperHeader

\maketitle

\input{supplementary/S2_experiments}
