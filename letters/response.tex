% \reviewer{N} starts a reviewer section and restarts comment numbering, so comments come out as Comment N.1, N.2, ... matching the \resp{RN.k}{...} margin tags in the revised manuscript.
\newcounter{reviewer}
\newcounter{comment}
\newcommand{\commentprefix}{}
\newcommand{\reviewer}[1]{%
  \setcounter{reviewer}{#1}\setcounter{comment}{0}%

  \section*{Response to Reviewer~#1}}
\newcommand{\associateeditor}{%
  \setcounter{comment}{0}\renewcommand{\commentprefix}{AE}%
  \section*{Response to the Associate Editor}}

\newenvironment{rcomment}{%
  \refstepcounter{comment}\par\medskip\noindent
  \textbf{Comment \commentprefix.\thecomment:}\itshape\ }{\par}
\newenvironment{answer}{%
  \par\smallskip\noindent\textbf{Response:}\ }{\par\medskip}

% Quote revised manuscript text, in blue like \rev{} in the paper.
\newcommand{\quotepaper}[1]{\begin{quote}\color{revblue}#1\end{quote}}
